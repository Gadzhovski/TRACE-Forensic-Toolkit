# TRACE runtime dependencies.
#
# One file for every platform: the Windows-only packages carry environment
# markers, so `pip install -r requirements.txt` is correct on Windows, macOS
# and Linux alike. (Previously pycaw, comtypes and python-magic-bin were
# declared unconditionally, which made this file fail outright on Linux.)
#
# Lower bounds rather than exact pins, so the tool keeps working on newer
# Python versions as wheels appear. Python 3.10 or newer is required: every
# package here installs from a pre-built wheel on Windows, macOS and Linux,
# so no compiler is needed anywhere.

# --- GUI -------------------------------------------------------------------
# Addons supplies QtCharts, QtSvg, QtMultimedia and QtPrintSupport, all used.
PySide6>=6.5.2

# --- Forensic image access -------------------------------------------------
# Pre-built wheels for Windows (x64, x86, ARM64), macOS (Apple Silicon and
# Intel) and Linux (x86_64, aarch64; glibc and musl) on Python 3.10+. These
# are the first releases that ship them; earlier ones were source-only and
# needed a C/C++ toolchain on every platform. The Sleuth Kit and libewf are
# built into the wheels, so no system packages are needed either.
pytsk3>=20260520
libewf-python>=20260924

# --- Evidence parsing ------------------------------------------------------
python-registry>=1.3.1   # Windows registry hives
PyMuPDF>=1.24.3          # PDF rendering, validation and metadata; 1.24.3
                         # introduced the `pymupdf` module name (`fitz` is
                         # deprecated and slated for removal)
Pillow>=11.3.0           # image decoding and EXIF; 11.3 is the first whose
                         # wheels decode AVIF, which the Application tab shows
chardet>=5.2.0           # text encoding detection
olefile>=0.46            # legacy Office (.doc/.xls/.ppt): authors and encryption
py7zr>=1.0.0             # 7z archives: browsing, and spotting encrypted ones.
                         # Wheels for every platform; resolves to 1.0 on
                         # Python 3.9 and on Intel macOS. LGPL-2.1+.
requests>=2.31.0         # VirusTotal API

# --- libmagic (file type from content) ------------------------------------
# python-magic is only a binding; libmagic is a C library. Windows: the -bin
# variant bundles the DLL. macOS: pylibmagic bundles libmagic and its
# database as a wheel (arm64 + x86_64, any Python 3), so no Homebrew; TRACE
# imports it before python-magic (trace_app/__init__.py). Pinned exactly: it
# is unmaintained since 2023, and CI checks it is the library actually
# loaded. Linux: libmagic1 from apt, needed alongside Qt's libraries anyway.
python-magic-bin>=0.4.14 ; sys_platform == 'win32'
python-magic>=0.4.27 ; sys_platform != 'win32'
pylibmagic==0.5.0 ; sys_platform == 'darwin'

# --- Windows-only audio endpoint volume control ----------------------------
pycaw>=20230407 ; sys_platform == 'win32'
comtypes>=1.2.0 ; sys_platform == 'win32'

# --- Tests (python -m pytest) -----------------------------------------------
# Image handling and the UI are tested on Windows, macOS and Linux in CI; the
# same suite runs locally. Not needed to run TRACE.
pytest>=8.0

# --- Building a standalone executable (build_exe.py) -----------------------
# Not needed to run TRACE from source; included so one install covers both.
pyinstaller>=6.0
